! NAT DIA for VPN2 Norway and VPN2 Auckland prefix 

viptela-policy:policy
 data-policy _VPN2_DIAv1-VPN2
  vpn-list VPN2
    sequence 1
     match
      source-data-prefix-list VPN2-ALL-Prefixes
      destination-data-prefix-list VPN2-ALL-Prefixes
     !
     action accept
     !
    !
    sequence 11
     match
      source-data-prefix-list DIA-Norway
     !
     action accept
      nat use-vpn 0
      nat fallback
     !
    !
    sequence 21
     match
      source-data-prefix-list DIA-Auckland
     !
     action accept
      nat use-vpn 0
      nat fallback
     !
    !
  default-action accept
 !
 control-policy Service-Chaining-VPN1wDIA
    sequence 1
     match route
      vpn-list VPN2
      prefix-list _AnyIpv4PrefixList
     !
     action accept
     !
    !
    sequence 11
     match route
      vpn-list VPN1
      site-list HQ-SITE
      prefix-list _AnyIpv4PrefixList
     !
     action accept
     !
    !
    sequence 21
     match route
      site-list SPOKE-SITES
      vpn-list VPN1
      prefix-list _AnyIpv4PrefixList
     !
     action accept
      set
       service  FW vpn 1 tloc-list HQ-TLOC
      !
     !
    !
  default-action accept
 !
 lists
  data-prefix-list DIA-Auckland
   ip-prefix 192.168.20.0/24 
  !
  data-prefix-list DIA-Norway
   ip-prefix 192.168.31.0/24 
  !
  data-prefix-list VPN2-ALL-Prefixes
   ip-prefix 192.168.0.0/16 
  !
  site-list HQ-SITE
   site-id 10 
  !
  site-list SPOKE-SITES
   site-id 20 
   site-id 30 
   site-id 40 
   site-id 50 
   site-id 60 
  !
  tloc-list HQ-TLOC
   tloc 10.10.10.10 color biz-internet encap ipsec 
   tloc 10.10.10.10 color mpls encap ipsec 
  !
  vpn-list VPN1
   vpn 1 
  !
  vpn-list VPN2
   vpn 2 
  !
  prefix-list _AnyIpv4PrefixList
   ip-prefix 0.0.0.0/0 le 32 
  !
 !
!
apply-policy
 site-list SPOKE-SITES
  data-policy _VPN2_DIAv1-VPN2 from-service
  control-policy Service-Chaining-VPN1wDIA out
 !
!
