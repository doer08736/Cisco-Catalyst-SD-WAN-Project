! Hub-N-Spoke VPN1 (with FW), VPN2 Full Mesh (DIA with Application Visibility) for Norway

viptela-policy:policy
 data-policy _VPN2_DIAv3-VPN2-AppVisibility
  vpn-list VPN2
    sequence 1
     match
      source-data-prefix-list VPN2-ALL-Prefixes-Exact
      destination-data-prefix-list VPN2-ALL-Prefixes-Exact
     !
     action accept
     !
    !
    sequence 11
     match
      source-data-prefix-list VPN2-ALL-Prefixes-Exact
      destination-data-prefix-list RFC1918
     !
     action drop
      log
     !
    !
    sequence 21
     match
      source-data-prefix-list DIA-Norway
      app-list Trusted-DIA-App-Norway
     !
     action accept
      nat use-vpn 0
      nat fallback
     !
    !
    sequence 31
     match
      source-data-prefix-list DIA-Auckland
     !
     action accept
      nat use-vpn 0
      nat fallback
     !
    !
  default-action accept
 !
 control-policy Service-Chaining-VPN1wDIAv3
    sequence 1
     match route
      vpn-list VPN2
      prefix-list _AnyIpv4PrefixList
     !
     action accept
     !
    !
    sequence 11
     match route
      vpn-list VPN1
      site-list HQ-SITE
      prefix-list _AnyIpv4PrefixList
     !
     action accept
     !
    !
    sequence 21
     match route
      site-list SPOKE-SITES
      vpn-list VPN1
      prefix-list _AnyIpv4PrefixList
     !
     action accept
      set
       service  FW vpn 1 tloc-list HQ-TLOC
      !
     !
    !
  default-action accept
 !
